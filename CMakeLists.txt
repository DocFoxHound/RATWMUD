cmake_minimum_required(VERSION 3.16)
project(RATWMUDCore LANGUAGES CXX)

option(RATW_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(ratw_core Core/RatwWorld.cpp Core/RatwAuthoring.cpp Core/RatwTravel.cpp Core/RatwCalendar.cpp Core/RatwSociety.cpp Core/RatwAging.cpp Core/RatwAppearance.cpp Core/RatwResidents.cpp Core/RatwPg.cpp Core/RatwBonds.cpp Core/RatwCareers.cpp Core/RatwRoads.cpp Core/RatwJsonDoc.cpp Core/RatwWire.cpp Core/RatwSocialCore.cpp Core/RatwCheckpoint.cpp Core/RatwDbStore.cpp Core/RatwSystemLibs.cpp Core/RatwAccountsCore.cpp Core/RatwMind.cpp Core/RatwSections.cpp Core/RatwMotionCore.cpp Core/RatwGame.cpp Core/RatwDirector.cpp Core/RatwWeb.cpp)
find_package(Threads REQUIRED)
target_link_libraries(ratw_core PUBLIC ${CMAKE_DL_LIBS} Threads::Threads)
target_include_directories(ratw_core PUBLIC Core)
if(MSVC)
  target_compile_options(ratw_core PRIVATE /W4)
else()
  target_compile_options(ratw_core PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
  if(RATW_SANITIZERS)
    target_compile_options(ratw_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ratw_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
