cmake_minimum_required(VERSION 3.16)
project(RATWMUDCore LANGUAGES CXX)

option(RATW_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(ratw_core Source/RATWMUD/Core/RatwWorld.cpp Source/RATWMUD/Core/RatwAuthoring.cpp Source/RATWMUD/Core/RatwTravel.cpp Source/RATWMUD/Core/RatwCalendar.cpp Source/RATWMUD/Core/RatwSociety.cpp Source/RATWMUD/Core/RatwAging.cpp Source/RATWMUD/Core/RatwAppearance.cpp)
target_include_directories(ratw_core PUBLIC Source/RATWMUD/Core)
if(MSVC)
  target_compile_options(ratw_core PRIVATE /W4)
else()
  target_compile_options(ratw_core PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
  if(RATW_SANITIZERS)
    target_compile_options(ratw_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ratw_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()

enable_testing()
add_executable(world_tests Tests/world_tests.cpp)
target_link_libraries(world_tests PRIVATE ratw_core)
target_compile_definitions(world_tests PRIVATE RATW_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME world_behaviors COMMAND world_tests)
add_executable(authoring_tests Tests/authoring_tests.cpp)
target_link_libraries(authoring_tests PRIVATE ratw_core)
add_test(NAME authoring_behaviors COMMAND authoring_tests)
add_executable(pace_tests Tests/pace_tests.cpp)
target_link_libraries(pace_tests PRIVATE ratw_core)
add_test(NAME pace_behaviors COMMAND pace_tests)
add_executable(travel_tests Tests/travel_tests.cpp)

add_test(NAME travel_behaviors COMMAND travel_tests)
add_executable(weather_tests Tests/weather_tests.cpp)
target_link_libraries(weather_tests PRIVATE ratw_core)
add_test(NAME weather_behaviors COMMAND weather_tests)
add_executable(lighting_tests Tests/lighting_tests.cpp)
target_link_libraries(lighting_tests PRIVATE ratw_core)
add_test(NAME lighting_behaviors COMMAND lighting_tests)
add_executable(calendar_tests Tests/calendar_tests.cpp)
target_link_libraries(calendar_tests PRIVATE ratw_core)
add_test(NAME calendar_behaviors COMMAND calendar_tests)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/society_tests.cpp")
  add_executable(society_tests Tests/society_tests.cpp)
  target_link_libraries(society_tests PRIVATE ratw_core)
  add_test(NAME society_behaviors COMMAND society_tests)
endif()
add_executable(aging_tests Tests/aging_tests.cpp)
target_link_libraries(aging_tests PRIVATE ratw_core)
add_test(NAME aging_behaviors COMMAND aging_tests)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Source/RATWMUD/Runtime/RatwSocialCore.cpp")
  add_library(ratw_social Source/RATWMUD/Runtime/RatwSocialCore.cpp)
  target_include_directories(ratw_social PUBLIC Source/RATWMUD/Runtime)
