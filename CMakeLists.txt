cmake_minimum_required(VERSION 3.16)
project(RATWMUDCore LANGUAGES CXX)

option(RATW_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(ratw_core Source/RATWMUD/Core/RatwWorld.cpp)
target_include_directories(ratw_core PUBLIC Source/RATWMUD/Core)
if(MSVC)
  target_compile_options(ratw_core PRIVATE /W4)
else()
  target_compile_options(ratw_core PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
  if(RATW_SANITIZERS)
    target_compile_options(ratw_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ratw_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()

enable_testing()
add_executable(world_tests Tests/world_tests.cpp)
target_link_libraries(world_tests PRIVATE ratw_core)
target_compile_definitions(world_tests PRIVATE RATW_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME world_behaviors COMMAND world_tests)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Source/RATWMUD/Runtime/RatwSocialCore.cpp")
  add_library(ratw_social Source/RATWMUD/Runtime/RatwSocialCore.cpp)
  target_include_directories(ratw_social PUBLIC Source/RATWMUD/Runtime)
  target_link_libraries(ratw_social PUBLIC ratw_core)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/social_tests.cpp")
    add_executable(social_tests Tests/social_tests.cpp)
    target_link_libraries(social_tests PRIVATE ratw_social)
    add_test(NAME social_behaviors COMMAND social_tests)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/RuntimeCoreTests.cpp")
    add_executable(runtime_core_tests Tests/RuntimeCoreTests.cpp)
    target_link_libraries(runtime_core_tests PRIVATE ratw_social)
    add_test(NAME runtime_core_behaviors COMMAND runtime_core_tests)
  endif()
endif()
